# TypeScript Types

Recast provides comprehensive TypeScript support with intelligent type inference and utility types for building type-safe component libraries.

## Core Types

### RecastClsProps

Adds a typed `cls` prop for components that use nested styling (object-based base classes).

```tsx
type RecastClsProps<T extends string> = {
  cls?: { [K in T]?: string }
}
```

**Usage:**

```tsx
import { RecastClsProps } from '@rpxl/recast'

interface CardProps extends RecastClsProps<'root' | 'header' | 'content' | 'footer'> {
  title?: string
  children: React.ReactNode
}

function CardPrimitive({ cls, title, children }: CardProps) {
  return (
    <div className={cls?.root}>
      <div className={cls?.header}>{title}</div>
      <div className={cls?.content}>{children}</div>
      <div className={cls?.footer}>Footer</div>
    </div>
  )
}

// TypeScript enforces the correct cls shape:
// cls?: { root?: string, header?: string, content?: string, footer?: string }
```

### MergeFn

Function type for custom class name merging.

```tsx
type MergeFn = (classes: string | string[], className?: string) => string
```

**Usage:**

```tsx
import { twMerge } from 'tailwind-merge'
import clsx from 'clsx'

// Both functions match the MergeFn type
const customMerge: MergeFn = (classes, className) => {
  return `${classes} ${className || ''}`.trim()
}

recast.configure({
  mergeFn: twMerge,    // ✅ Valid
  mergeFn: clsx,       // ✅ Valid  
  mergeFn: customMerge // ✅ Valid
})
```

### ClassNameRecord

Type for nested class name objects.

```tsx
type ClassNameRecord = Record<string, string | string[]>
```

**Usage:**

```tsx
// Used internally for nested component styling
const nestedClasses: ClassNameRecord = {
  root: "rounded-lg border",
  header: ["px-6", "py-4", "border-b"],
  content: "px-6 py-4"
}
```

## Style Configuration Types

### RecastStyles

The complete type for style configuration objects passed to `recast.styles()`.

```tsx
interface RecastStyles<V, M, T> {
  base?: string | ClassNameRecord | string[]
  variants?: V
  modifiers?: M  
  conditionals?: Conditional<V, M>[]
  defaults?: {
    variants?: Partial<ExtractVariantProps<V>>
    modifiers?: Array<keyof M>
  }
}
```

**Example:**

```tsx
// TypeScript infers the complete configuration type
const buttonConfig: RecastStyles<
  { variant: { primary: string, secondary: string } },
  { disabled: string, loading: string },
  { cls?: ClassNameRecord }
> = {
  base: "px-4 py-2 rounded",
  variants: {
    variant: {
      primary: "bg-blue-500 text-white",
      secondary: "bg-gray-200 text-gray-900"
    }
  },
  modifiers: {
    disabled: "opacity-50 cursor-not-allowed",
    loading: "cursor-wait"
  },
  conditionals: [
    {
      variants: { variant: "primary" },
      modifiers: ["loading"],
      className: "bg-blue-400"
    }
  ],
  defaults: {
    variants: { variant: "primary" },
    modifiers: ["disabled"]
  }
}
```

### Conditional

Type for conditional styling rules.

```tsx
interface Conditional<V, M> {
  variants?: Partial<ExtractVariantProps<V>>
  modifiers?: Array<keyof M>
  className: string | ClassNameRecord
}
```

## Extracted Prop Types

### ExtractVariantProps

Extracts variant prop types from a variant configuration.

```tsx
type ExtractVariantProps<V> = {
  [K in keyof V]?: keyof V[K]
}
```

**Example:**

```tsx
const variants = {
  size: { sm: "text-sm", md: "text-base", lg: "text-lg" },
  variant: { primary: "bg-blue-500", secondary: "bg-gray-500" }
}

type VariantProps = ExtractVariantProps<typeof variants>
// Result: {
//   size?: "sm" | "md" | "lg"
//   variant?: "primary" | "secondary"
// }
```

### ExtractModifierProps

Extracts modifier prop types from a modifier configuration.

```tsx
type ExtractModifierProps<M> = {
  [K in keyof M]?: boolean
}
```

**Example:**

```tsx
const modifiers = {
  disabled: "opacity-50",
  loading: "cursor-wait",
  fullWidth: "w-full"
}

type ModifierProps = ExtractModifierProps<typeof modifiers>
// Result: {
//   disabled?: boolean
//   loading?: boolean
//   fullWidth?: boolean
// }
```

## Component Prop Types

### Extracting Component Props

Get the complete prop types from a styled component:

```tsx
const buttonStyles = recast.styles({
  variants: {
    variant: { primary: "...", secondary: "..." },
    size: { sm: "...", md: "...", lg: "..." }
  },
  modifiers: {
    disabled: "...",
    loading: "..."
  }
})

const Button = buttonStyles(ButtonPrimitive)

// Extract complete prop types
type ButtonProps = ComponentProps<typeof Button>
// Result: {
//   variant?: "primary" | "secondary"
//   size?: "sm" | "md" | "lg"
//   disabled?: boolean
//   loading?: boolean
//   className?: string
//   // + all ButtonPrimitive props
// }
```

### Custom Component Interfaces

Create custom interfaces that extend component props:

```tsx
// Restrict to specific variants
interface PrimaryButtonProps extends Omit<ButtonProps, 'variant'> {
  variant?: 'primary'
}

// Make certain props required
interface RequiredSizeButtonProps extends ButtonProps {
  size: NonNullable<ButtonProps['size']> // size is now required
}

// Add custom props
interface IconButtonProps extends ButtonProps {
  icon: React.ReactNode
  iconPosition?: 'left' | 'right'
}
```

## Advanced Type Patterns

### Conditional Prop Types

Create props that depend on other prop values:

```tsx
// Button can have an href only when variant is "link"
type ConditionalButtonProps = ButtonProps & (
  | { variant: 'link'; href: string }
  | { variant?: Exclude<ButtonProps['variant'], 'link'>; href?: never }
)

// Usage
<ConditionalButton variant="link" href="/home" /> // ✅ Valid
<ConditionalButton variant="primary" /> // ✅ Valid
<ConditionalButton variant="primary" href="/home" /> // ❌ TypeScript error
```

### Generic Style Components

Create reusable style utilities with generic types:

```tsx
// Generic card component that accepts different content types
interface GenericCardProps<T> extends CardProps {
  data: T
  render: (data: T) => React.ReactNode
}

function GenericCard<T>({ data, render, ...cardProps }: GenericCardProps<T>) {
  return (
    <Card {...cardProps}>
      {render(data)}
    </Card>
  )
}

// Usage with full type safety
<GenericCard 
  data={{ name: "John", age: 30 }}
  render={(user) => <span>{user.name} is {user.age} years old</span>}
  variant="primary"
  size="lg"
/>
```

### Polymorphic Components

Create components that can render as different HTML elements:

```tsx
type AsProps<T extends React.ElementType> = {
  as?: T
} & React.ComponentPropsWithoutRef<T>

type PolymorphicButtonProps<T extends React.ElementType = 'button'> = 
  ButtonProps & AsProps<T>

function PolymorphicButton<T extends React.ElementType = 'button'>({
  as,
  ...props
}: PolymorphicButtonProps<T>) {
  const Component = as || 'button'
  return <Button {...props} />
}

// Usage
<PolymorphicButton>Default button</PolymorphicButton>
<PolymorphicButton as="a" href="/link">Link button</PolymorphicButton>
<PolymorphicButton as={Link} to="/route">Router link</PolymorphicButton>
```

## Utility Types

### Deep Readonly

Make style configurations immutable:

```tsx
type DeepReadonly<T> = {
  readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P]
}

const readonlyButtonConfig: DeepReadonly<typeof buttonConfig> = buttonConfig
// All properties are now readonly at every level
```

### Style Object Type Guards

Type guards for runtime type checking:

```tsx
function isVariantProp<V>(
  prop: string, 
  variants: V
): prop is keyof V {
  return prop in variants
}

function isModifierProp<M>(
  prop: string,
  modifiers: M
): prop is keyof M {
  return prop in modifiers
}

// Usage in dynamic prop handling
function handleDynamicProps(props: Record<string, unknown>) {
  Object.entries(props).forEach(([key, value]) => {
    if (isVariantProp(key, buttonConfig.variants)) {
      // TypeScript knows key is a valid variant
      console.log(`Variant ${key} set to ${value}`)
    } else if (isModifierProp(key, buttonConfig.modifiers)) {
      // TypeScript knows key is a valid modifier
      console.log(`Modifier ${key} is ${value}`)
    }
  })
}
```

## Type Debugging

### Inspecting Inferred Types

Use TypeScript utilities to inspect complex inferred types:

```tsx
// See the exact type TypeScript infers
type InspectButtonProps = ComponentProps<typeof Button>

// Break down complex types
type VariantTypes = ExtractVariantProps<typeof buttonConfig.variants>
type ModifierTypes = ExtractModifierProps<typeof buttonConfig.modifiers>

// Use TypeScript's built-in utilities
type RequiredButtonProps = Required<ButtonProps>
type PartialButtonProps = Partial<ButtonProps>
type PickedProps = Pick<ButtonProps, 'variant' | 'size'>
type OmittedProps = Omit<ButtonProps, 'disabled' | 'loading'>
```

### Type Testing

Create type-level tests to ensure your interfaces are correct:

```tsx
// Type-level tests (these should compile without errors)
const _testVariantTypes: ExtractVariantProps<typeof buttonConfig.variants> = {
  variant: "primary", // ✅ Should work
  size: "lg"         // ✅ Should work
  // variant: "invalid" // ❌ Should cause TypeScript error
}

const _testModifierTypes: ExtractModifierProps<typeof buttonConfig.modifiers> = {
  disabled: true,    // ✅ Should work  
  loading: false,    // ✅ Should work
  // disabled: "yes" // ❌ Should cause TypeScript error
}
```

## Best Practices

<AccordionGroup>
  <Accordion title="Use RecastClsProps for nested components">
    Always type your nested component props with RecastClsProps:
    
    ```tsx
    // ✅ Good: Properly typed nested component
    interface CardProps extends RecastClsProps<'root' | 'header' | 'content'> {
      title: string
    }
    
    // ❌ Bad: Untyped cls prop
    interface CardProps {
      cls?: any
      title: string
    }
    ```
  </Accordion>
  
  <Accordion title="Extract and reuse prop types">
    Create reusable type definitions for consistency:
    
    ```tsx
    // ✅ Good: Extracted and reusable
    export type ButtonProps = ComponentProps<typeof Button>
    export type CardProps = ComponentProps<typeof Card>
    
    // Use in other interfaces
    interface FormProps {
      submitButton?: ButtonProps
      cancelButton?: ButtonProps
    }
    ```
  </Accordion>
  
  <Accordion title="Provide generic constraints">
    Use generic constraints for flexible but safe APIs:
    
    ```tsx
    // ✅ Good: Constrained generic
    function createStyledComponent<T extends React.ElementType>(
      element: T,
      styles: RecastStylesObject
    ): (props: ComponentProps<T>) => JSX.Element
    
    // ❌ Bad: Unconstrained generic
    function createStyledComponent<T>(element: T, styles: any): any
    ```
  </Accordion>
</AccordionGroup>
