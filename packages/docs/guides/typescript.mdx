# TypeScript Guide

Master TypeScript with Recast to build type-safe, maintainable component libraries. This guide covers advanced patterns, utility types, and best practices for leveraging TypeScript's power with Recast.

<Note>
  This guide assumes familiarity with TypeScript fundamentals. We'll focus on advanced patterns specific to component library development with Recast.
</Note>

## Core TypeScript Integration

Recast provides excellent TypeScript support out of the box with automatic type inference and utility types.

### Automatic Type Inference

Recast automatically infers types from your style configurations:

```tsx Type Inference Example
import { recast } from '@rpxl/recast'

const buttonStyles = recast.styles({
  variants: {
    variant: {
      primary: "bg-blue-500 text-white",
      secondary: "bg-gray-200 text-gray-900"
    },
    size: {
      sm: "px-3 py-1.5 text-sm",
      md: "px-4 py-2",
      lg: "px-6 py-3 text-lg"
    }
  },
  modifiers: {
    disabled: "opacity-50 cursor-not-allowed",
    loading: "cursor-wait",
    fullWidth: "w-full"
  }
})

// TypeScript automatically infers this type:
type InferredButtonProps = {
  variant?: "primary" | "secondary"
  size?: "sm" | "md" | "lg"
  disabled?: boolean
  loading?: boolean
  fullWidth?: boolean
  className?: string
  // + all props from the wrapped component
}

const Button = buttonStyles(ButtonPrimitive)

// Usage with full type safety
<Button 
  variant="primary"    // ✅ Autocomplete works
  size="lg"           // ✅ Valid
  disabled={true}     // ✅ Valid
  // variant="invalid" // ❌ TypeScript error
/>
```

## Advanced Type Patterns

### Conditional Props

Create props that depend on other prop values:

```tsx Conditional Props
// Button can have an href only when variant is "link"
type ConditionalButtonProps = ButtonProps & (
  | { variant: 'link'; href: string; target?: string }
  | { variant?: Exclude<ButtonProps['variant'], 'link'>; href?: never; target?: never }
)

function ConditionalButton(props: ConditionalButtonProps) {
  if (props.variant === 'link') {
    // TypeScript knows href is available here
    return <a href={props.href} target={props.target}>{props.children}</a>
  }
  
  // TypeScript knows href is not available here
  return <Button {...props} />
}

// Usage
<ConditionalButton variant="link" href="/home" target="_blank" /> // ✅ Valid
<ConditionalButton variant="primary" /> // ✅ Valid
<ConditionalButton variant="primary" href="/home" /> // ❌ TypeScript error
```

### Polymorphic Components

Create components that can render as different elements:

```tsx Polymorphic Components
import { ElementType, ComponentPropsWithoutRef } from 'react'

type AsProps<T extends ElementType> = {
  as?: T
}

type PolymorphicRef<T extends ElementType> = ComponentPropsWithoutRef<T>['ref']

type PolymorphicComponentProps<T extends ElementType, P = {}> = 
  P & AsProps<T> & Omit<ComponentPropsWithoutRef<T>, 'as' | keyof P>

type PolymorphicComponent<T extends ElementType, P = {}> = <C extends ElementType = T>(
  props: PolymorphicComponentProps<C, P> & { ref?: PolymorphicRef<C> }
) => JSX.Element

// Create polymorphic button
type ButtonOwnProps = {
  variant?: 'primary' | 'secondary'
  size?: 'sm' | 'md' | 'lg'
}

const PolymorphicButton: PolymorphicComponent<'button', ButtonOwnProps> = forwardRef(
  <T extends ElementType = 'button'>(
    { as, variant, size, ...props }: PolymorphicComponentProps<T, ButtonOwnProps>,
    ref: PolymorphicRef<T>
  ) => {
    const Component = as || 'button'
    const classes = buttonStyles.extract({ variant, size })
    
    return (
      <Component
        ref={ref}
        className={classes}
        {...props}
      />
    )
  }
)

// Usage with full type safety
<PolymorphicButton>Default button</PolymorphicButton>
<PolymorphicButton as="a" href="/link">Link button</PolymorphicButton>
<PolymorphicButton as={Link} to="/route">Router link</PolymorphicButton>
<PolymorphicButton as="div" role="button">Div button</PolymorphicButton>
```

### Generic Style Components

Create reusable style utilities with generics:

```tsx Generic Components
// Generic card component with typed data
interface GenericCardProps<T> extends CardProps {
  data: T
  renderContent: (data: T) => React.ReactNode
  renderActions?: (data: T) => React.ReactNode
}

function GenericCard<T>({ 
  data, 
  renderContent, 
  renderActions, 
  ...cardProps 
}: GenericCardProps<T>) {
  return (
    <Card {...cardProps}>
      <Card.Content>
        {renderContent(data)}
      </Card.Content>
      {renderActions && (
        <Card.Footer>
          {renderActions(data)}
        </Card.Footer>
      )}
    </Card>
  )
}

// Usage with full type safety
interface User {
  id: string
  name: string
  email: string
  role: 'admin' | 'user'
}

<GenericCard<User>
  data={user}
  variant="primary"
  renderContent={(user) => (
    <div>
      <h3>{user.name}</h3>
      <p>{user.email}</p>
      <Badge variant={user.role === 'admin' ? 'primary' : 'secondary'}>
        {user.role}
      </Badge>
    </div>
  )}
  renderActions={(user) => (
    <Button 
      size="sm" 
      onClick={() => editUser(user.id)}
    >
      Edit {user.name}
    </Button>
  )}
/>
```

### Discriminated Unions

Use discriminated unions for complex component variants:

```tsx Discriminated Unions
// Alert with different structures based on variant
type AlertProps = 
  | {
      variant: 'success' | 'info' | 'warning'
      title: string
      description?: string
      action?: ButtonProps
    }
  | {
      variant: 'error'
      title: string
      description: string // Required for errors
      action: ButtonProps   // Required for errors
      retry?: () => void
    }
  | {
      variant: 'loading'
      title: string
      description?: never   // Not allowed for loading
      action?: never        // Not allowed for loading
      progress?: number
    }

function Alert(props: AlertProps) {
  if (props.variant === 'error') {
    // TypeScript knows description and action are available
    return (
      <Card variant="error">
        <Card.Header>
          <Card.Title>{props.title}</Card.Title>
          <Card.Description>{props.description}</Card.Description>
        </Card.Header>
        <Card.Footer>
          <Button {...props.action} />
          {props.retry && (
            <Button variant="outline" onClick={props.retry}>
              Retry
            </Button>
          )}
        </Card.Footer>
      </Card>
    )
  }
  
  if (props.variant === 'loading') {
    // TypeScript knows progress is available, description/action are not
    return (
      <Card variant="info">
        <Card.Content>
          <div className="flex items-center space-x-3">
            <Spinner />
            <div>
              <p className="font-medium">{props.title}</p>
              {props.progress && (
                <div className="w-full bg-gray-200 rounded-full h-2">
                  <div 
                    className="bg-blue-600 h-2 rounded-full transition-all"
                    style={{ width: `${props.progress}%` }}
                  />
                </div>
              )}
            </div>
          </div>
        </Card.Content>
      </Card>
    )
  }
  
  // For success, info, warning variants
  return (
    <Card variant={props.variant}>
      <Card.Header>
        <Card.Title>{props.title}</Card.Title>
        {props.description && (
          <Card.Description>{props.description}</Card.Description>
        )}
      </Card.Header>
      {props.action && (
        <Card.Footer>
          <Button {...props.action} />
        </Card.Footer>
      )}
    </Card>
  )
}

// Usage with type safety
<Alert variant="success" title="Success!" />
<Alert variant="error" title="Error" description="Something went wrong" action={{ children: "Fix" }} />
<Alert variant="loading" title="Loading..." progress={75} />
```

## Utility Types and Helpers

### Style Configuration Types

Create reusable types for style configurations:

```tsx Style Configuration Types
// Extract style configuration types
type StyleVariants<T> = T extends { variants: infer V } ? V : never
type StyleModifiers<T> = T extends { modifiers: infer M } ? M : never

// Get variant keys
type VariantKeys<T> = keyof StyleVariants<T>
type ModifierKeys<T> = keyof StyleModifiers<T>

// Example usage
type ButtonVariants = StyleVariants<typeof buttonStyles._config>
type ButtonModifiers = StyleModifiers<typeof buttonStyles._config>

// Create utility to extract specific variant types
type ExtractVariant<T, K extends VariantKeys<T>> = 
  StyleVariants<T>[K] extends Record<infer U, any> ? U : never

type ButtonVariantOptions = ExtractVariant<typeof buttonStyles._config, 'variant'>
// Result: "primary" | "secondary"

type ButtonSizeOptions = ExtractVariant<typeof buttonStyles._config, 'size'>  
// Result: "sm" | "md" | "lg"
```

### Composable Style Types

Type utilities for style composition:

```tsx Composable Types
// Merge multiple style configurations
type MergeStyleConfigs<T extends readonly any[]> = T extends readonly [
  infer First,
  ...infer Rest
] 
  ? First & MergeStyleConfigs<Rest>
  : {}

// Extract props from composed styles
type ComposedProps<T extends readonly any[]> = ComponentProps<
  ReturnType<typeof recast.compose<T>>
>

// Example
const composedStyles = recast.compose([layoutStyles, colorStyles, sizeStyles])
type ComposedButtonProps = ComposedProps<typeof composedStyles>

// Create style builder pattern
class StyleBuilder<T = {}> {
  private config: T

  constructor(config: T = {} as T) {
    this.config = config
  }

  addVariants<V>(variants: V): StyleBuilder<T & { variants: V }> {
    return new StyleBuilder({ ...this.config, variants })
  }

  addModifiers<M>(modifiers: M): StyleBuilder<T & { modifiers: M }> {
    return new StyleBuilder({ ...this.config, modifiers })
  }

  build() {
    return recast.styles(this.config)
  }
}

// Usage
const dynamicButtonStyles = new StyleBuilder()
  .addVariants({
    variant: { primary: "bg-blue-500", secondary: "bg-gray-500" },
    size: { sm: "px-2 py-1", md: "px-4 py-2" }
  })
  .addModifiers({
    disabled: "opacity-50",
    loading: "cursor-wait"
  })
  .build()
```

### Form Type Utilities

Advanced form typing patterns:

```tsx Form Type Utilities
// Create form data types from field configurations
type FormField<T = any> = {
  name: string
  label: string
  type: 'text' | 'email' | 'password' | 'select' | 'checkbox' | 'textarea'
  required?: boolean
  validation?: (value: T) => string | null
  options?: Array<{ label: string; value: T }>
}

type FormConfig = {
  [K: string]: FormField
}

// Extract form data type from configuration
type FormData<T extends FormConfig> = {
  [K in keyof T]: T[K] extends FormField<infer U> 
    ? T[K]['required'] extends true 
      ? U 
      : U | undefined
    : never
}

// Example form configuration
const userFormConfig = {
  name: {
    name: 'name',
    label: 'Full Name', 
    type: 'text' as const,
    required: true,
    validation: (value: string) => value.length < 2 ? 'Name too short' : null
  },
  email: {
    name: 'email',
    label: 'Email',
    type: 'email' as const,
    required: true,
    validation: (value: string) => !/\S+@\S+\.\S+/.test(value) ? 'Invalid email' : null
  },
  newsletter: {
    name: 'newsletter',
    label: 'Subscribe to newsletter',
    type: 'checkbox' as const,
    required: false
  }
} as const

// TypeScript infers the exact form data type
type UserFormData = FormData<typeof userFormConfig>
// Result: {
//   name: string
//   email: string  
//   newsletter: boolean | undefined
// }

// Type-safe form component
function TypedForm<T extends FormConfig>({
  config,
  onSubmit
}: {
  config: T
  onSubmit: (data: FormData<T>) => void
}) {
  const [data, setData] = useState<Partial<FormData<T>>>({})
  const [errors, setErrors] = useState<Partial<Record<keyof T, string>>>({})
  
  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    
    // Type-safe validation
    const newErrors: Partial<Record<keyof T, string>> = {}
    
    for (const [key, field] of Object.entries(config)) {
      const value = data[key as keyof T]
      
      if (field.required && !value) {
        newErrors[key as keyof T] = `${field.label} is required`
        continue
      }
      
      if (value && field.validation) {
        const error = field.validation(value)
        if (error) {
          newErrors[key as keyof T] = error
        }
      }
    }
    
    if (Object.keys(newErrors).length === 0) {
      onSubmit(data as FormData<T>) // Safe cast after validation
    } else {
      setErrors(newErrors)
    }
  }
  
  return (
    <form onSubmit={handleSubmit}>
      {Object.entries(config).map(([key, field]) => (
        <FormField key={key}>
          <Label required={field.required}>{field.label}</Label>
          <Input
            type={field.type}
            value={data[key as keyof T] || ''}
            onChange={(e) => setData({ ...data, [key]: e.target.value })}
            error={errors[key as keyof T]}
          />
          {errors[key as keyof T] && (
            <FieldMessage variant="error">
              {errors[key as keyof T]}
            </FieldMessage>
          )}
        </FormField>
      ))}
      <Button type="submit">Submit</Button>
    </form>
  )
}

// Usage with full type safety
<TypedForm 
  config={userFormConfig}
  onSubmit={(data) => {
    // data is fully typed as UserFormData
    console.log(data.name)      // string
    console.log(data.email)     // string
    console.log(data.newsletter) // boolean | undefined
  }}
/>
```

### Runtime Type Validation

Combine TypeScript with runtime validation:

```tsx Runtime Validation
import { z } from 'zod'

// Create Zod schemas that match your TypeScript types
const ButtonPropsSchema = z.object({
  variant: z.enum(['primary', 'secondary']).optional(),
  size: z.enum(['sm', 'md', 'lg']).optional(),
  disabled: z.boolean().optional(),
  loading: z.boolean().optional(),
  fullWidth: z.boolean().optional(),
  className: z.string().optional()
})

// Create a type-safe component wrapper
function createValidatedComponent<P>(
  Component: React.ComponentType<P>,
  schema: z.ZodSchema<P>
) {
  return function ValidatedComponent(props: P) {
    if (process.env.NODE_ENV === 'development') {
      const result = schema.safeParse(props)
      if (!result.success) {
        console.error('Invalid props:', result.error.format())
      }
    }
    
    return <Component {...props} />
  }
}

// Create validated button
const ValidatedButton = createValidatedComponent(Button, ButtonPropsSchema)

// Usage - props are validated at runtime in development
<ValidatedButton variant="primary" size="lg" />
```

## Best Practices

### Type Organization

Organize your types for maintainability:

```tsx Type Organization
// types/components.ts
export type ButtonProps = ComponentProps<typeof Button>
export type CardProps = ComponentProps<typeof Card>
export type InputProps = ComponentProps<typeof Input>

// types/forms.ts
export type FormFieldProps = ComponentProps<typeof FormField>
export type FormData<T extends FormConfig> = { /* ... */ }

// types/layouts.ts
export type GridProps = ComponentProps<typeof Grid>
export type ContainerProps = ComponentProps<typeof Container>

// types/utils.ts
export type StyleVariants<T> = T extends { variants: infer V } ? V : never
export type StyleModifiers<T> = T extends { modifiers: infer M } ? M : never

// types/index.ts
export * from './components'
export * from './forms'
export * from './layouts'
export * from './utils'
```

### Performance Considerations

TypeScript performance tips for large component libraries:

```tsx Performance Tips
// ✅ Good: Use specific imports
import type { ButtonProps } from './Button'

// ❌ Avoid: Importing everything
import type * as Components from './components'

// ✅ Good: Prefer type-only imports when possible
import type { ComponentProps } from 'react'

// ✅ Good: Use const assertions for better inference
const buttonConfig = {
  variants: {
    variant: { primary: "...", secondary: "..." }
  }
} as const

// ✅ Good: Extract types early in development
type ButtonStyleConfig = typeof buttonConfig
```
