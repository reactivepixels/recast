# Best Practices

Learn proven patterns and strategies for building maintainable, scalable design systems with Recast. This guide covers architecture, team collaboration, testing, and long-term maintenance.

<Note>
  These practices are based on real-world experience building and maintaining large-scale design systems. Adapt them to fit your team's needs and constraints.
</Note>

## Project Architecture

Structure your project for maintainability and scalability:

## Component Design Patterns

### Primitive Pattern

Start with unstyled primitives for maximum flexibility:

```tsx Primitive Pattern
// components/button/button-primitive.tsx
import { forwardRef, ButtonHTMLAttributes } from 'react'
import { RecastClsProps } from '@rpxl/recast'

interface ButtonPrimitiveProps 
  extends ButtonHTMLAttributes<HTMLButtonElement>,
          RecastClsProps<'root'> {
  children: React.ReactNode
}

export const ButtonPrimitive = forwardRef<HTMLButtonElement, ButtonPrimitiveProps>(
  ({ cls, children, ...props }, ref) => {
    return (
      <button ref={ref} className={cls?.root} {...props}>
        {children}
      </button>
    )
  }
)

ButtonPrimitive.displayName = 'ButtonPrimitive'

// components/button/button.tsx
import { buttonStyles } from './button.styles'
import { ButtonPrimitive } from './button-primitive'

export const Button = buttonStyles(ButtonPrimitive)
export type ButtonProps = ComponentProps<typeof Button>
```

### Provider Pattern

Share common configuration across components:

```tsx Provider Pattern
// providers/theme-provider.tsx
import { createContext, useContext } from 'react'

interface ThemeConfig {
  colors: typeof colors
  spacing: typeof spacing
  borderRadius: 'none' | 'sm' | 'md' | 'lg'
  animations: boolean
}

const ThemeContext = createContext<ThemeConfig | null>(null)

export function ThemeProvider({ 
  children, 
  theme 
}: { 
  children: React.ReactNode
  theme: ThemeConfig 
}) {
  // Configure Recast with theme
  useEffect(() => {
    recast.configure({
      mergeFn: twMerge,
      performance: {
        enableMonitoring: process.env.NODE_ENV === 'development'
      }
    })
  }, [])

  return (
    <ThemeContext.Provider value={theme}>
      {children}
    </ThemeContext.Provider>
  )
}

export const useTheme = () => {
  const context = useContext(ThemeContext)
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider')
  }
  return context
}

// Use in components
function ThemedButton({ variant, ...props }) {
  const theme = useTheme()
  
  const themedButtonStyles = useMemo(() => 
    recast.styles({
      variants: {
        variant: {
          primary: `bg-${theme.colors.primary} text-white`,
          secondary: `bg-${theme.colors.secondary} text-gray-900`
        }
      }
    }), [theme])

  const ThemedButtonComponent = themedButtonStyles(ButtonPrimitive)
  return <ThemedButtonComponent variant={variant} {...props} />
}
```

### Responsive Design System

Build responsive components systematically:

```tsx Responsive System
// utils/responsive.ts
export const breakpoints = {
  sm: '640px',
  md: '768px', 
  lg: '1024px',
  xl: '1280px'
} as const

export const createResponsiveStyles = (base: string, responsive: Partial<Record<keyof typeof breakpoints, string>>) => {
  const classes = [base]
  
  Object.entries(responsive).forEach(([breakpoint, value]) => {
    classes.push(`${breakpoint}:${value}`)
  })
  
  return classes.join(' ')
}

// components/grid/grid.styles.ts
export const gridStyles = recast.styles({
  base: {
    root: 'grid'
  },
  
  variants: {
    cols: {
      1: { root: 'grid-cols-1' },
      2: { root: 'grid-cols-2' },
      3: { root: 'grid-cols-3' },
      4: { root: 'grid-cols-4' }
    },
    gap: {
      none: { root: 'gap-0' },
      sm: { root: 'gap-2' },
      md: { root: 'gap-4' },
      lg: { root: 'gap-6' }
    }
  },
  
  // Responsive conditionals
  conditionals: [
    {
      variants: { cols: 1 },
      className: { root: 'sm:grid-cols-1 md:grid-cols-2 lg:grid-cols-3' }
    }
  ]
})

// Enhanced responsive grid component
interface ResponsiveGridProps extends GridProps {
  responsive?: {
    sm?: number
    md?: number
    lg?: number
    xl?: number
  }
}

export function ResponsiveGrid({ cols, responsive, className, ...props }: ResponsiveGridProps) {
  const responsiveClasses = responsive ? 
    createResponsiveStyles(
      `grid-cols-${cols || 1}`,
      Object.fromEntries(
        Object.entries(responsive).map(([bp, colCount]) => [bp, `grid-cols-${colCount}`])
      )
    ) : ''

  const GridComponent = gridStyles('div')
  
  return (
    <GridComponent
      cols={cols}
      className={`${className || ''} ${responsiveClasses}`.trim()}
      {...props}
    />
  )
}
```

## Testing Strategies

### Component Testing

Test components and their style variants:

```tsx Component Testing
// components/button/button.test.tsx
import { render, screen } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { Button } from './button'
import { buttonStyles } from './button.styles'

describe('Button', () => {
  it('renders correctly', () => {
    render(<Button>Click me</Button>)
    expect(screen.getByRole('button')).toBeInTheDocument()
  })
  
  it('applies variant styles correctly', () => {
    const { rerender } = render(<Button variant="primary">Primary</Button>)
    
    const button = screen.getByRole('button')
    expect(button).toHaveClass('bg-blue-500')
    
    rerender(<Button variant="secondary">Secondary</Button>)
    expect(button).toHaveClass('bg-gray-200')
  })
  
  it('handles size variants', () => {
    const { rerender } = render(<Button size="sm">Small</Button>)
    
    const button = screen.getByRole('button')
    expect(button).toHaveClass('h-8')
    
    rerender(<Button size="lg">Large</Button>)
    expect(button).toHaveClass('h-10')
  })
  
  it('supports modifiers', () => {
    render(<Button disabled>Disabled</Button>)
    
    const button = screen.getByRole('button')
    expect(button).toBeDisabled()
    expect(button).toHaveClass('opacity-50')
  })
  
  it('handles click events', async () => {
    const handleClick = jest.fn()
    render(<Button onClick={handleClick}>Click me</Button>)
    
    await userEvent.click(screen.getByRole('button'))
    expect(handleClick).toHaveBeenCalledTimes(1)
  })
})
```

## Performance Best Practices

### Optimization Guidelines

Follow these guidelines for optimal performance:

```tsx Performance Guidelines
// ✅ Good: Create styles outside components
const buttonStyles = recast.styles({
  // Style definition
})

function MyComponent() {
  return <Button>Good</Button>
}

// ❌ Bad: Creating styles inside components
function MyComponent() {
  const buttonStyles = recast.styles({
    // Recreated on every render!
  })
  return <Button>Bad</Button>
}

// ✅ Good: Stable prop values
const BUTTON_VARIANTS = {
  save: 'primary',
  cancel: 'secondary'
} as const

function Form({ onSave, onCancel }) {
  return (
    <div>
      <Button variant={BUTTON_VARIANTS.save} onClick={onSave}>
        Save
      </Button>
      <Button variant={BUTTON_VARIANTS.cancel} onClick={onCancel}>
        Cancel
      </Button>
    </div>
  )
}

// ✅ Good: Memoize expensive computations
function ComplexComponent({ theme, features }) {
  const dynamicStyles = useMemo(() => 
    recast.styles({
      conditionals: features.map(feature => ({
        modifiers: [feature.enabled],
        className: feature.styles
      }))
    }), [features])

  return <div>Complex component</div>
}

// ✅ Good: Extract styles for reuse
const commonButtonClasses = buttonStyles.extract({ variant: 'primary' })

function MultipleButtons() {
  return (
    <div>
      {/* Use extracted classes instead of computing each time */}
      <button className={commonButtonClasses}>Button 1</button>
      <button className={commonButtonClasses}>Button 2</button>
      <button className={commonButtonClasses}>Button 3</button>
    </div>
  )
}
```

```tsx Migration Patterns
// migrations/v2-to-v3.ts
export const migrationGuide = {

    {

    },

      automated: false,

    }
  ],

    {
      component: 'OldButton',

    }
  ]

// Automated migration script
export function runMigration(sourceCode: string): string {
  return sourceCode
    .replace(/size="small"/g, 'size="sm"')
    .replace(/size="medium"/g, 'size="md"')
    .replace(/size="large"/g, 'size="lg"')
    .replace(/elevation="high"/g, 'elevated')
    .replace(/elevation="low"/g, '')
    .replace(/<OldButton/g, '<Button')
}

// Gradual migration strategy
export function createMigrationWrapper<T>(
  OldComponent: React.ComponentType<T>,
  NewComponent: React.ComponentType<T>,
  migrationOptions: {
    warnInDev?: boolean
    trackUsage?: boolean
  } = {}
) {
  return function MigrationWrapper(props: T) {
    if (process.env.NODE_ENV === 'development' && migrationOptions.warnInDev) {
      console.warn('Component is deprecated. Please migrate to the new version.')
    }
    
    if (migrationOptions.trackUsage) {
      // Track usage for analytics
      trackDeprecatedComponentUsage(OldComponent.displayName)
    }
    
    // Use old component during migration period
    return <OldComponent {...props} />
  }
}
```

## Monitoring and Analytics

### Usage Analytics

Track component usage in production:

```tsx Usage Analytics
// utils/analytics.ts
interface ComponentUsageEvent {
  component: string
  variant?: string
  props: Record<string, any>
  timestamp: number
  userId?: string
}

class ComponentAnalytics {
  private events: ComponentUsageEvent[] = []
  
  track(component: string, props: Record<string, any>) {
    if (process.env.NODE_ENV === 'production') {
      this.events.push({
        component,
        props,
        timestamp: Date.now(),
        userId: getCurrentUserId()
      })
    }
  }
  
  flush() {
    if (this.events.length > 0) {
      sendAnalytics(this.events)
      this.events = []
    }
  }
  
  getUsageReport() {
    const usage = new Map<string, number>()
    
    this.events.forEach(event => {
      const key = `${event.component}:${JSON.stringify(event.props)}`
      usage.set(key, (usage.get(key) || 0) + 1)
    })
    
    return Object.fromEntries(usage)
  }
}

export const analytics = new ComponentAnalytics()

// Enhanced component wrapper with analytics
export function withAnalytics<P>(
  Component: React.ComponentType<P>,
  componentName: string
) {
  return function AnalyticsWrapper(props: P) {
    useEffect(() => {
      analytics.track(componentName, props)
    }, [props])
    
    return <Component {...props} />
  }
}

// Usage
export const AnalyticsButton = withAnalytics(Button, 'Button')
```
