# Card Component

This example demonstrates how to build a sophisticated card component with nested styling, showcasing Recast's power for complex UI components with multiple parts.

<Note>
  This example shows advanced patterns including nested object styling, composition, conditional styling, and flexible layout options.
</Note>

## What we'll build

A flexible card component with multiple styled parts and comprehensive customization options:

<CodeGroup>

## Step 1: Create the primitives

Start with clean, unstyled primitives for each card part:

<CodeGroup>
```tsx CardPrimitive.tsx
import React, { HTMLAttributes, forwardRef } from 'react'
import { RecastClsProps } from '@rpxl/recast'

interface CardPrimitiveProps 
  extends HTMLAttributes<HTMLDivElement>,
          RecastClsProps<'root' | 'header' | 'title' | 'description' | 'content' | 'footer'> {
  children: React.ReactNode
}

export const CardPrimitive = forwardRef<HTMLDivElement, CardPrimitiveProps>(
  ({ cls, children, ...props }, ref) => {
    return (
      <div ref={ref} className={cls?.root} {...props}>
        {children}
      </div>
    )
  }
)

CardPrimitive.displayName = 'CardPrimitive'
```

### Conditional styles

Complex styling logic for specific combinations:

```tsx conditionalStyles.ts
export const cardConditionalStyles = recast.styles({
  conditionals: [
    // Large elevated cards get extra shadow
    {
      variants: { size: "lg" },
      modifiers: ["elevated"],
      className: {
        root: "shadow-2xl"
      }
    },
    
    // XL cards get even more dramatic shadows when elevated
    {
      variants: { size: "xl" },
      modifiers: ["elevated"],
      className: {
        root: "shadow-2xl shadow-gray-500/20"
      }
    },
    
    // Hoverable elevated cards have enhanced hover effects
    {
      modifiers: ["hoverable", "elevated"],
      className: {
        root: "hover:shadow-2xl hover:-translate-y-1"
      }
    },
    
    // Dense cards without headers need different spacing
    {
      modifiers: ["dense"],
      className: {
        content: "pt-4"
      }
    },
    
    // Borderless elevated cards need special treatment
    {
      modifiers: ["borderless", "elevated"],
      className: {
        root: "shadow-xl"
      }
    },
    
    // Primary variant with hoverable gets special hover color
    {
      variants: { variant: "primary" },
      modifiers: ["hoverable"],
      className: {
        root: "hover:bg-blue-100"
      }
    },
    
    // Error variant gets pulsing animation when hoverable
    {
      variants: { variant: "error" },
      modifiers: ["hoverable"],
      className: {
        root: "hover:animate-pulse"
      }
    },
    
    // Small dense cards get minimal padding
    {
      variants: { size: "sm" },
      modifiers: ["dense"],
      className: {
        header: "px-3 py-1.5",
        content: "px-3 py-1.5",
        footer: "px-3 py-1.5"
      }
    }
  ]
})
```

## Step 3: Compose the card styles

Bring all style modules together:

```tsx Card.tsx
import { recast } from '@rpxl/recast'
import { 
  CardPrimitive,
  CardHeaderPrimitive,
  CardTitlePrimitive,
  CardDescriptionPrimitive,
  CardContentPrimitive,
  CardFooterPrimitive
} from './CardPrimitives'
import {
  cardFoundationStyles,
  cardLayoutStyles,
  cardVisualStyles,
  cardEnhancementStyles,
  cardConditionalStyles
} from './cardStyles'

// Compose main card styles
const cardStyles = recast.compose([
  cardFoundationStyles,
  cardLayoutStyles,
  cardVisualStyles,
  cardEnhancementStyles,
  cardConditionalStyles
])

// Create styled components
export const Card = cardStyles(CardPrimitive)

// Create simple styled subcomponents (these don't need complex styling)
const headerStyles = recast.styles({
  base: { root: "" } // Let parent card control styling
})

const titleStyles = recast.styles({
  base: { root: "" }
})

const descriptionStyles = recast.styles({
  base: { root: "" }
})

const contentStyles = recast.styles({
  base: { root: "" }
})

const footerStyles = recast.styles({
  base: { root: "" }
})

// Attach subcomponents to main Card
Card.Header = headerStyles(CardHeaderPrimitive)
Card.Title = titleStyles(CardTitlePrimitive)
Card.Description = descriptionStyles(CardDescriptionPrimitive)
Card.Content = contentStyles(CardContentPrimitive)
Card.Footer = footerStyles(CardFooterPrimitive)

// Export types
export type CardProps = ComponentProps<typeof Card>
```

## Step 5: Style extraction for advanced use cases

Extract computed styles for integration with other systems:

```tsx Style Extraction
// Get computed classes for specific combinations
const primaryCardClasses = Card.extract({
  variant: "primary",
  size: "lg", 
  elevated: true,
  hoverable: true
})

// Use with external libraries
function MotionCard({ children, ...props }) {
  const cardClasses = Card.extract(props)
  
  return (
    <motion.div
      className={cardClasses.root}
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      whileHover={{ scale: 1.02 }}
    >
      {children}
    </motion.div>
  )
}

// Server-side rendering
function generateCardHTML(props: CardProps, content: string) {
  const classes = Card.extract(props)
  return `
    <div class="${classes.root}">
      <div class="${classes.header}">
        <h3 class="${classes.title}">Title</h3>
        <p class="${classes.description}">Description</p>
      </div>
      <div class="${classes.content}">
        ${content}
      </div>
    </div>
  `
}
```

## Testing the card component

Test individual style modules and the composed result:

```tsx Card.test.tsx
import { describe, it, expect } from 'vitest'
import { render } from '@testing-library/react'
import { Card } from './Card'
import { cardLayoutStyles, cardVisualStyles } from './cardStyles'

describe('Card Component', () => {
  it('applies correct layout classes for different sizes', () => {
    expect(cardLayoutStyles.extract({ size: 'sm' }))
      .toEqual(expect.objectContaining({
        root: expect.stringContaining('max-w-sm'),
        header: expect.stringContaining('px-4 py-3')
      }))
  })
  
  it('applies correct visual styles for variants', () => {
    const primaryClasses = cardVisualStyles.extract({ variant: 'primary' })
    expect(primaryClasses.root).toContain('border-blue-200')
    expect(primaryClasses.title).toContain('text-blue-900')
  })
  
  it('renders with composed styles', () => {
    const { container } = render(
      <Card variant="primary" size="lg" elevated>
        <Card.Header>
          <Card.Title>Test Title</Card.Title>
        </Card.Header>
        <Card.Content>Test content</Card.Content>
      </Card>
    )
    
    const card = container.querySelector('div')
    expect(card).toHaveClass('max-w-lg') // from size
    expect(card).toHaveClass('border-blue-200') // from variant
    expect(card).toHaveClass('shadow-lg') // from elevated
  })
  
  it('applies conditional styling correctly', () => {
    const classes = Card.extract({ 
      variant: "primary",
      modifiers: ["hoverable", "elevated"],
      size: "xl"
    })
    
    expect(classes.root).toContain('hover:shadow-2xl')
    expect(classes.root).toContain('shadow-2xl')
  })
})
```

## Key patterns demonstrated

This card example showcases several important patterns:

<AccordionGroup>
  <Accordion title="Nested object styling">
    The card uses object-based base classes to style multiple parts of a complex component with a single style definition.
  </Accordion>
  
  <Accordion title="Subcomponent composition">
    Card.Header, Card.Title, etc. are attached as properties, creating a clean component API.
  </Accordion>
  
  <Accordion title="Multi-layer composition">
    Foundation, layout, visual, enhancement, and conditional styles are composed in layers.
  </Accordion>
  
  <Accordion title="Semantic variants">
    Success, warning, error variants communicate purpose and provide appropriate styling.
  </Accordion>
  
  <Accordion title="Complex conditionals">
    Multiple conditional rules handle edge cases and sophisticated styling combinations.
  </Accordion>
</AccordionGroup>
