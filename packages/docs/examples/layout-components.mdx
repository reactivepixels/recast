# Layout Components

Learn how to create powerful layout systems that handle complex responsive designs while maintaining clean, semantic markup. This example demonstrates grid systems, flexible containers, and adaptive layouts.

<Note>
  This example shows how to build layout primitives that can be composed to create any design while maintaining design system consistency.
</Note>

## Step 1: Build container system

Foundation for consistent page layouts:

```tsx Container.tsx
import { recast } from '@rpxl/recast'

const containerStyles = recast.styles({
  base: {
    root: "mx-auto px-4 sm:px-6 lg:px-8"
  },
  
  variants: {
    size: {
      sm: {
        root: "max-w-sm"
      },
      md: {
        root: "max-w-md"
      },
      lg: {
        root: "max-w-4xl"
      },
      xl: {
        root: "max-w-6xl"
      },
      "2xl": {
        root: "max-w-7xl"
      },
      full: {
        root: "max-w-full"
      },
      none: {
        root: "max-w-none"
      }
    }
  },
  
  modifiers: {
    fluid: {
      root: "px-0"
    },
    centered: {
      root: "text-center"
    }
  },
  
  defaults: {
    variants: { size: "xl" }
  }
})

interface ContainerProps extends HTMLAttributes<HTMLDivElement> {
  size?: "sm" | "md" | "lg" | "xl" | "2xl" | "full" | "none"
  fluid?: boolean
  centered?: boolean
  as?: React.ElementType
}

export const Container = forwardRef<HTMLDivElement, ContainerProps>(
  ({ size, fluid, centered, as = "div", ...props }, ref) => {
    const ContainerComponent = containerStyles(as)
    
    return (
      <ContainerComponent
        ref={ref}
        size={size}
        fluid={fluid}
        centered={centered}
        {...props}
      />
    )
  }
)
```

## Step 2: Create flexible grid system

Responsive grid with powerful customization:

```tsx Grid.tsx
const gridStyles = recast.styles({
  base: {
    root: "grid"
  },
  
  variants: {
    cols: {
      1: { root: "grid-cols-1" },
      2: { root: "grid-cols-2" },
      3: { root: "grid-cols-3" },
      4: { root: "grid-cols-4" },
      5: { root: "grid-cols-5" },
      6: { root: "grid-cols-6" },
      8: { root: "grid-cols-8" },
      12: { root: "grid-cols-12" }
    },
    gap: {
      none: { root: "gap-0" },
      xs: { root: "gap-1" },
      sm: { root: "gap-2" },
      md: { root: "gap-4" },
      lg: { root: "gap-6" },
      xl: { root: "gap-8" }
    },
    align: {
      start: { root: "items-start" },
      center: { root: "items-center" },
      end: { root: "items-end" },
      stretch: { root: "items-stretch" }
    },
    justify: {
      start: { root: "justify-items-start" },
      center: { root: "justify-items-center" },
      end: { root: "justify-items-end" },
      stretch: { root: "justify-items-stretch" }
    }
  },
  
  defaults: {
    variants: { cols: 1, gap: "md" }
  }
})

// Enhanced grid with responsive breakpoints
const responsiveGridStyles = recast.styles({
  conditionals: [
    // Responsive breakpoint classes would be added here
    // This is a simplified example - in practice you'd generate these
  ]
})

const composedGridStyles = recast.compose([
  gridStyles,
  responsiveGridStyles
])

interface GridProps extends HTMLAttributes<HTMLDivElement> {
  cols?: 1 | 2 | 3 | 4 | 5 | 6 | 8 | 12
  gap?: "none" | "xs" | "sm" | "md" | "lg" | "xl"
  align?: "start" | "center" | "end" | "stretch"
  justify?: "start" | "center" | "end" | "stretch"
  responsive?: {
    sm?: number
    md?: number
    lg?: number
    xl?: number
  }
}

export const Grid = forwardRef<HTMLDivElement, GridProps>(
  ({ cols, gap, align, justify, responsive, className, ...props }, ref) => {
    // Build responsive classes
    const responsiveClasses = responsive ? Object.entries(responsive)
      .map(([breakpoint, cols]) => {
        const prefix = breakpoint === 'sm' ? 'sm:' : 
                     breakpoint === 'md' ? 'md:' :
                     breakpoint === 'lg' ? 'lg:' : 'xl:'
        return `${prefix}grid-cols-${cols}`
      }).join(' ') : ''
    
    const GridComponent = composedGridStyles('div')
    
    return (
      <GridComponent
        ref={ref}
        cols={cols}
        gap={gap}
        align={align}
        justify={justify}
        className={`${className || ''} ${responsiveClasses}`.trim()}
        {...props}
      />
    )
  }
)
```

### App Layout

Common application layout pattern:

```tsx AppLayout.tsx
const appLayoutStyles = recast.styles({
  base: {
    root: "min-h-screen bg-gray-50",
    header: "bg-white border-b border-gray-200 sticky top-0 z-40",
    sidebar: "bg-white border-r border-gray-200 fixed inset-y-0 left-0 z-30",
    main: "transition-all duration-200",
    content: "max-w-full mx-auto py-6 px-4 sm:px-6 lg:px-8"
  },
  
  variants: {
    sidebarWidth: {
      sm: {
        sidebar: "w-48",
        main: "ml-48"
      },
      md: {
        sidebar: "w-64",
        main: "ml-64"
      },
      lg: {
        sidebar: "w-80",
        main: "ml-80"
      }
    }
  },
  
  modifiers: {
    sidebarCollapsed: {
      sidebar: "w-16",
      main: "ml-16"
    },
    sidebarHidden: {
      sidebar: "hidden",
      main: "ml-0"
    },
    fullWidth: {
      content: "max-w-none"
    }
  },
  
  defaults: {
    variants: { sidebarWidth: "md" }
  }
})

interface AppLayoutProps {
  children: React.ReactNode
  sidebarWidth?: "sm" | "md" | "lg"
  sidebarCollapsed?: boolean
  sidebarHidden?: boolean
  fullWidth?: boolean
}

export const AppLayout = forwardRef<HTMLDivElement, AppLayoutProps>(
  ({ children, sidebarWidth, sidebarCollapsed, sidebarHidden, fullWidth }, ref) => {
    const AppLayoutComponent = appLayoutStyles(({ cls }) => (
      <div ref={ref} className={cls?.root}>
        {children}
      </div>
    ))
    
    return (
      <AppLayoutComponent
        sidebarWidth={sidebarWidth}
        sidebarCollapsed={sidebarCollapsed}
        sidebarHidden={sidebarHidden}
        fullWidth={fullWidth}
      />
    )
  }
)

// Subcomponents
const headerStyles = recast.styles({
  base: { root: "" }
})

const sidebarStyles = recast.styles({
  base: { root: "" }
})

const mainStyles = recast.styles({
  base: { root: "" }
})

const contentStyles = recast.styles({
  base: { root: "" }
})

AppLayout.Header = headerStyles(({ cls, ...props }) => (
  <header className={cls?.root} {...props} />
))

AppLayout.Sidebar = sidebarStyles(({ cls, ...props }) => (
  <aside className={cls?.root} {...props} />
))

AppLayout.Main = mainStyles(({ cls, ...props }) => (
  <main className={cls?.root} {...props} />
))

AppLayout.Content = contentStyles(({ cls, ...props }) => (
  <div className={cls?.root} {...props} />
))
```

```tsx ArticleLayout.tsx
      <Grid cols={4} gap="xl" responsive={{ sm: 1, lg: 4 }}>
        {/* Article Content */}
        <GridItem span={3}>
          <Stack spacing="xl">
            {/* Article Header */}
            <Stack spacing="md">
              <h1 className="text-4xl font-bold leading-tight">
                {article.title}
              </h1>
              <p className="text-xl text-gray-600">
                {article.excerpt}
              </p>
              <Flex gap="md" align="center" className="text-sm text-gray-500">
                <img 
                  src={article.author.avatar} 
                  alt={article.author.name}
                  className="w-8 h-8 rounded-full"
                />
                <span>{article.author.name}</span>
                <span>•</span>
                <time>{formatDate(article.publishedAt)}</time>
                <span>•</span>
                <span>{article.readTime} min read</span>
              </Flex>
            </Stack>

            {/* Article Image */}
            {article.featuredImage && (
              <img 
                src={article.featuredImage} 
                alt={article.title}
                className="w-full rounded-lg"
              />
            )}

            {/* Article Body */}
            <div className="prose prose-lg max-w-none">
              <MDXContent content={article.content} />
            </div>

            {/* Article Footer */}
            <Stack spacing="md" className="border-t pt-8">
              <Flex wrap gap="sm">
                {article.tags.map(tag => (
                  <Badge key={tag} variant="secondary">
                    {tag}
                  </Badge>
                ))}
              </Flex>
              <Flex justify="between" align="center">
                <ShareButtons article={article} />
                <Flex gap="md">
                  <Button variant="outline" size="sm">
                    Previous
                  </Button>
                  <Button variant="outline" size="sm">
                    Next
                  </Button>
                </Flex>
              </Flex>
            </Stack>
          </Stack>
        </GridItem>

        {/* Sidebar */}
        <GridItem>
          <Stack spacing="lg" className="sticky top-8">
            {/* Table of Contents */}
            <Card>
              <Card.Header>
                <Card.Title>Table of Contents</Card.Title>
              </Card.Header>
              <Card.Content>
                <TableOfContents headings={article.headings} />
              </Card.Content>
            </Card>

            {/* Author Bio */}
            <Card>
              <Card.Header>
                <Card.Title>About the Author</Card.Title>
              </Card.Header>
              <Card.Content>
                <Stack spacing="sm">
                  <Flex gap="sm" align="center">
                    <img 
                      src={article.author.avatar}
                      alt={article.author.name}
                      className="w-12 h-12 rounded-full"
                    />
                    <Stack spacing="xs">
                      <h4 className="font-semibold">{article.author.name}</h4>
                      <p className="text-sm text-gray-600">{article.author.role}</p>
                    </Stack>
                  </Flex>
                  <p className="text-sm">{article.author.bio}</p>
                </Stack>
              </Card.Content>
            </Card>

            {/* Related Articles */}
            <Card>
              <Card.Header>
                <Card.Title>Related Articles</Card.Title>
              </Card.Header>
              <Card.Content>
                <Stack spacing="sm">
                  {article.relatedArticles.map(related => (
                    <RelatedArticleCard key={related.id} article={related} />
                  ))}
                </Stack>
              </Card.Content>
            </Card>
          </Stack>
        </GridItem>
      </Grid>
    </Container>
  )
}
```
