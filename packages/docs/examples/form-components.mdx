# Form Components

Learn how to create a cohesive set of form components that work together seamlessly. This example demonstrates consistent styling patterns, validation states, and accessibility features across all form elements.

<Note>
  This example shows how to build a design system of form components that share common patterns while maintaining individual customization options.
</Note>

## Step 2: Build specific input components

### Text Input

Enhanced text input with validation states:

```tsx Input.tsx
import { forwardRef, InputHTMLAttributes } from 'react'
import { recast } from '@rpxl/recast'
import { inputFoundationStyles } from './formStyles'

// Enhanced input with loading and validation
const inputEnhancementStyles = recast.styles({
  conditionals: [
    {
      modifiers: ["loading"],
      className: {
        root: "pr-8" // Make room for loading spinner
      }
    }
  ]
})

// Compose input styles
const inputStyles = recast.compose([
  inputFoundationStyles,
  inputEnhancementStyles
])

interface InputProps extends Omit<InputHTMLAttributes<HTMLInputElement>, 'size'> {
  size?: "sm" | "md" | "lg"
  error?: string | boolean
  success?: boolean
  loading?: boolean
}

export const Input = forwardRef<HTMLInputElement, InputProps>(
  ({ size, error, success, loading, className, ...props }, ref) => {
    const InputComponent = inputStyles(({ className: cls, ...inputProps }) => (
      <div className="relative">
        <input 
          ref={ref}
          className={cls}
          {...inputProps}
        />
        {loading && (
          <div className="absolute right-2 top-1/2 -translate-y-1/2">
            <LoadingSpinner size="sm" />
          </div>
        )}
      </div>
    ))
    
    return (
      <InputComponent
        size={size}
        error={!!error}
        success={success}
        loading={loading}
        className={className}
        {...props}
      />
    )
  }
)
```

### Textarea

Textarea with consistent styling and auto-resize:

```tsx Textarea.tsx
const textareaStyles = recast.compose([
  inputFoundationStyles,
  recast.styles({
    base: {
      root: "min-h-[80px] resize-y"
    },
    
    variants: {
      resize: {
        none: { root: "resize-none" },
        vertical: { root: "resize-y" },
        horizontal: { root: "resize-x" },
        both: { root: "resize" }
      }
    },
    
    modifiers: {
      autoResize: {
        root: "resize-none overflow-hidden"
      }
    },
    
    defaults: {
      variants: { resize: "vertical" }
    }
  })
])

interface TextareaProps extends TextareaHTMLAttributes<HTMLTextAreaElement> {
  size?: "sm" | "md" | "lg"
  resize?: "none" | "vertical" | "horizontal" | "both"
  autoResize?: boolean
  error?: string | boolean
  success?: boolean
}

export const Textarea = forwardRef<HTMLTextAreaElement, TextareaProps>(
  ({ size, resize, autoResize, error, success, ...props }, ref) => {
    const TextareaComponent = textareaStyles('textarea')
    
    return (
      <TextareaComponent
        ref={ref}
        size={size}
        resize={resize}
        autoResize={autoResize}
        error={!!error}
        success={success}
        {...props}
      />
    )
  }
)
```

```tsx Select.tsx
const selectStyles = recast.compose([
```

```tsx Checkbox.tsx
const checkboxStyles = recast.styles({

        input: "h-5 w-5",

      }
    }

    error: {

    },
    disabled: {
      root: "opacity-50 cursor-not-allowed",
      label: "cursor-not-allowed"
    }
  },
  
  defaults: {
    variants: { size: "md" }
  }
})

interface CheckboxProps extends Omit<InputHTMLAttributes<HTMLInputElement>, 'size' | 'type'> {
  size?: "sm" | "md" | "lg"
  error?: boolean
  children: React.ReactNode
}

export const Checkbox = forwardRef<HTMLInputElement, CheckboxProps>(
  ({ size, error, disabled, children, id, ...props }, ref) => {
    const CheckboxComponent = checkboxStyles(({ cls }) => (
      <div className={cls?.root}>
        <input
          ref={ref}
          type="checkbox"
          id={id}
          className={cls?.input}
          disabled={disabled}
          {...props}
        />
        <label htmlFor={id} className={cls?.label}>
          {children}
        </label>
      </div>
    ))
    
    return (
      <CheckboxComponent
        size={size}
        error={error}
        disabled={disabled}
      />
    )
  }
)
```

### FieldMessage

Helper text and validation messages:

```tsx FieldMessage.tsx
interface FieldMessageProps {
  variant?: "hint" | "error" | "success" | "warning" | "loading"
  children: React.ReactNode
  icon?: React.ReactNode
}

export const FieldMessage = forwardRef<HTMLDivElement, FieldMessageProps>(
  ({ variant, children, icon }, ref) => {
    const MessageComponent = fieldMessageStyles(({ cls }) => (
      <div ref={ref} className={cls?.root}>
        {icon && <span className="flex-shrink-0">{icon}</span>}
        <span>{children}</span>
      </div>
    ))
    
    return (
      <MessageComponent variant={variant} />
    )
  }
)
```

### Dynamic Form with Validation

Form with real-time validation and dynamic fields:

```tsx DynamicForm.tsx
function DynamicForm() {
  const [fields, setFields] = useState([
    { id: 1, name: '', email: '', role: 'member' }
  ])
  const [validationState, setValidationState] = useState({})

  const addField = () => {
    setFields([...fields, { 
      id: Date.now(), 
      name: '', 
      email: '', 
      role: 'member' 
    }])
  }

  const removeField = (id) => {
    setFields(fields.filter(field => field.id !== id))
  }

  const updateField = (id, updates) => {
    setFields(fields.map(field => 
      field.id === id ? { ...field, ...updates } : field
    ))
  }

  return (
    <Form className="space-y-6">
      <div className="space-y-4">
        <h3 className="text-lg font-semibold">Team Members</h3>
        
        {fields.map((field, index) => (
          <Card key={field.id} variant="default" className="p-4">
            <div className="flex items-start justify-between mb-4">
              <h4 className="font-medium">Member {index + 1}</h4>
              {fields.length > 1 && (
                <Button
                  variant="ghost"
                  size="sm"
                  onClick={() => removeField(field.id)}
                >
                  Remove
                </Button>
              )}
            </div>
            
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <FormField>
                <Label required>Name</Label>
                <Input
                  value={field.name}
                  onChange={(e) => updateField(field.id, { name: e.target.value })}
                  placeholder="Full name"
                  error={validationState[`name_${field.id}`]}
                />
              </FormField>

              <FormField>
                <Label required>Email</Label>
                <Input
                  type="email"
                  value={field.email}
                  onChange={(e) => updateField(field.id, { email: e.target.value })}
                  placeholder="email@company.com"
                  error={validationState[`email_${field.id}`]}
                />
              </FormField>
            </div>

            <FormField>
              <Label>Role</Label>
              <Select
                value={field.role}
                onChange={(e) => updateField(field.id, { role: e.target.value })}
              >
                <option value="member">Member</option>
                <option value="admin">Admin</option>
                <option value="owner">Owner</option>
              </Select>
            </FormField>
          </Card>
        ))}
      </div>

      <Button variant="outline" onClick={addField} fullWidth>
        + Add Another Member
      </Button>

      <Button variant="primary" size="lg" fullWidth>
        Create Team
      </Button>
    </Form>
  )
}
```
